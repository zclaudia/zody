import {readFile,writeFile} from 'node:fs/promises';
import {gunzipSync} from 'node:zlib';
import {compareGraphs,compareLicenseEvidence} from '../../packages/desktop/scripts/lib/cua-driver-closure-comparison-graph.mjs';
import {parseCargoTree} from '../../packages/desktop/scripts/lib/cua-driver-closure-dependencies.mjs';
import {runClosureCommand} from '../../packages/desktop/scripts/lib/cua-driver-closure-command.mjs';
const r8=JSON.parse(gunzipSync(await readFile('packages/desktop/native/macos-cua-driver-closure/reports/2026-10-28-round8/report.json.gz')));
const r9=JSON.parse(await readFile('output/cua-driver-round9/controller-final-audit/report.json'));
const packages=d=>d.dependencies.graph.packages;
const graph=compareGraphs(packages(r8),packages(r9));
const licenses=compareLicenseEvidence({report:r8,packages:packages(r8)},{report:r9,packages:packages(r9)});
const config=JSON.parse(await readFile('output/cua-driver-round9/controller-final-native/commands.json'));
const args=['tree','--locked','--offline','--target','aarch64-apple-darwin','--package','cua-macos-visual-link-probe','--edges','normal,build','--prefix','none','--format','{p}|{f}'];
const command=config.environment.RUSTC.replace(/rustc$/,'cargo');
const probe=await runClosureCommand(command,args,{cwd:config.source,env:config.environment,timeoutMs:30000,maxOutputBytes:1048576});
await writeFile('output/cua-driver-round9/controller-final-native/consumer-tree.log',probe.stdout);
const parsed=parseCargoTree(probe.stdout);
const comparison={kind:'round9-controller-graph-comparison',scope:'Same visual root; narrower implementation, not full Driver/ABI equivalence. System frameworks are not Cargo graph packages.',graph,licenses,probe:{command,args,code:probe.code,...parsed},lock:r9.dependencies.lock,source:r9.source.counts};
await writeFile('output/cua-driver-round9/controller-checks/comparison.json',JSON.stringify(comparison,null,2)+'\n');
console.log(JSON.stringify({removed:graph.removed.map(x=>x.key),added:graph.added,featureChanged:graph.featureChanged,probePackages:parsed.packages.length,licenses:licenses.candidate}));
